```typescript
// src/Tool.ts:362
export type Tool<Input, Output, P = unknown> = {
  // === 身份标识 ===
  readonly name: string                    // 工具唯一名称，如 "Bash"
  aliases?: string[]                      // 别名列表，如 ["Shell", "Terminal"]
  searchHint?: string                     // 工具搜索关键词

  // === 模型接口 ===
  readonly inputSchema: AnyObject        // Zod 输入 schema
  outputSchema?: z.ZodType<unknown>    // Zod 输出 schema

  // prompt() 返回给模型的工具描述
  prompt(options: ToolPromptOptions): Promise<string>

  // description() 返回运行时描述（含动态信息）
  description(input: Input, options: ToolPromptOptions): Promise<string>

  // === 执行接口 ===
  call(
    args: Input,
    context: ToolUseContext,
    canUseTool: CanUseToolFn,
    parentMessage?: Message,
    onProgress?: (progress: Progress) => void
  ): Promise<ToolResult<Output>>

  // === 权限与安全 ===
  checkPermissions?(input: Input, context: ToolUseContext): Promise<PermissionResult>
  validateInput?(input: Input, context: ToolUseContext): Promise<ValidationResult>
  isReadOnly(input: Input): boolean                     // 是否只读操作
  isDestructive?(input: Input): boolean                  // 是否破坏性操作
  isConcurrencySafe(input: Input): boolean             // 是否可并发执行

  // === 可用性 ===
  isEnabled(): boolean                                  // 当前环境是否可用

  // === UI 相关 ===
  userFacingName(input: Input): string                  // 用户可见名称
  maxResultSizeChars: number                           // 结果最大字符数

  // === 中断行为 ===
  interruptBehavior?(): 'cancel' | 'block'              // 被中断时的行为

  // === MCP 标识 ===
  isMcp?: boolean                                      // 是否 MCP 工具
  mcpInfo?: { serverName: string; toolName: string }  // MCP 元信息
}
```

## 2. 工具注册表：getAllBaseTools()

**文件**：`src/tools.ts:50-150`

**功能**：注册所有内置工具，支持条件加载。

```typescript
// src/tools.ts:50
export function getAllBaseTools(): Tools {
  const tools: Tools = []

  // === 核心工具（始终加载）===
  tools.push(AgentTool)
  tools.push(TaskOutputTool)
  tools.push(BashTool)

  // === 搜索工具（条件加载）===
  if (hasEmbeddedSearchTools()) {
    // 跳过 GlobTool, GrepTool
  } else {
    tools.push(GlobTool)
    tools.push(GrepTool)
  }

  // === 文件操作工具 ===
  tools.push(ExitPlanModeV2Tool)
  tools.push(FileReadTool)
  tools.push(FileEditTool)
  tools.push(FileWriteTool)
  tools.push(NotebookEditTool)

  // === 网络工具 ===
  tools.push(WebFetchTool)
  tools.push(WebSearchTool)

  // === 任务工具 ===
  tools.push(TodoWriteTool)
  tools.push(TaskStopTool)
  tools.push(AskUserQuestionTool)

  // === Skill 和 Plan ===
  tools.push(SkillTool)
  tools.push(EnterPlanModeTool)

  // === 条件加载的工具 ===
  if (isTodoV2Enabled()) {
    tools.push(TaskCreateTool)
    tools.push(TaskGetTool)
    tools.push(TaskListTool)
  }

  if (isWorktreeModeEnabled()) {
    tools.push(EnterWorktreeTool)
    tools.push(ExitWorktreeTool)
  }

  // === Agent 间通信 ===
  tools.push(getSendMessageTool())

  // === Swarm 工具 ===
  if (isAgentSwarmsEnabled()) {
    tools.push(TeamCreateTool)
    tools.push(TeamDeleteTool)
  }

  // === MCP 资源工具 ===
  tools.push(BriefTool)
  tools.push(ListMcpResourcesTool)
  tools.push(ReadMcpResourceTool)

  // === Tool Search ===
  if (isToolSearchEnabledOptimistic()) {
    tools.push(ToolSearchTool)
  }

  return tools
}
```

## 3. 工具池组装：assembleToolPool()

**文件**：`src/tools.ts:200-280`

**功能**：合并内置工具和 MCP 工具，去重并排序。

```typescript
// src/tools.ts:200
export function assembleToolPool(
  permissionContext: PermissionContext,
  mcpTools: Tools = []
): Tools {
  // 1. 获取过滤后的内置工具
  const builtInTools = getTools(permissionContext)

  // 2. 过滤 MCP 工具（应用 deny 规则）
  const filteredMcpTools = filterToolsByDenyRules(mcpTools, permissionContext)

  // 3. 合并并去重（内置优先）
  const combined = [...builtInTools, ...filteredMcpTools]

  // 4. 按名称排序（为 prompt cache 稳定性）
  const sorted = combined.sort((a, b) => a.name.localeCompare(b.name))

  // 5. 去重（保留第一个）
  const deduplicated = uniqBy(sorted, 'name')

  return deduplicated
}
```

### 4.1 call() 方法结构

```typescript
// src/tools/BashTool/BashTool.tsx:100-200
async function call(
  args: { command: string },
  context: ToolUseContext,
  canUseTool: CanUseToolFn,
): Promise<ToolResult<string>> {
  const { command } = args

  // 1. 安全检查
  const securityResult = parseForSecurity(command)
  if (!securityResult.isSafe) {
    return { success: false, error: securityResult.error }
  }

  // 2. 读取/只读约束检查
  const readOnlyResult = checkReadOnlyConstraints(command, context)
  if (!readOnlyResult.allowed) {
    return { success: false, error: `Read-only constraint violated` }
  }

  // 3. 沙箱决策
  const useSandbox = shouldUseSandbox(command, context)

  // 4. 权限检查
  const permissionResult = await canUseTool(this, args, context)
  if (permissionResult.behavior === 'deny') {
    return { success: false, error: permissionResult.message }
  }

  // 5. 执行命令
  const execResult = await exec(command, {
    cwd: context.options.cwd || process.cwd(),
    timeout: getTimeoutForCommand(command),
    sandbox: useSandbox,
  })

  return { success: true, data: execResult.output }
}
```
