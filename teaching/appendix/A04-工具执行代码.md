## 3. 并发执行器：StreamingToolExecutor

**文件**：`src/services/tools/StreamingToolExecutor.ts:40-150`

**功能**：管理模型同时发出的多个 tool_use 的并发执行。

### 3.1 状态机

```
TrackedTool 状态转换：
  queued → executing → completed/yielded
                ↓
              errored → (触发 siblingAbortController.abort)
```

### 3.2 核心类结构

```typescript
// src/services/tools/StreamingToolExecutor.ts:40-80
class StreamingToolExecutor {
  private tools: Map<string, TrackedTool> = new Map()
  private siblingAbortController: AbortController = new AbortController()

  async addTool(toolUse: ToolUseBlock, ...): Promise<void> {
    const tool = new TrackedTool(toolUse, toolName, ...)
    this.tools.set(tool.id, tool)

    if (this.canExecuteTool(tool)) {
      this.executeTool(tool)
    }
  }

  private canExecuteTool(tool: TrackedTool): boolean {
    if (!tool.isConcurrencySafe()) {
      const hasRunningNonConcurrent = [...this.tools.values()]
        .some(t => t.isExecuting() && !t.isConcurrencySafe())
      if (hasRunningNonConcurrent) return false
    }
    return true
  }
}
```

### 3.3 兄弟错误取消机制

```typescript
// src/services/tools/StreamingToolExecutor.ts:150-180
if (tool.block.name === BASH_TOOL_NAME) {
  this.hasErrored = true
  this.siblingAbortController.abort('sibling_error')

  for (const [id, t] of this.tools) {
    if (t.isExecuting() && t.block.name === BASH_TOOL_NAME) {
      t.abortController.abort('sibling_error')
    }
  }
}
```

**设计意图**：Bash 命令通常有隐式依赖（如 `mkdir` 失败后 `cd` 无意义），其他类型的工具互相独立。

## 4. Hook 系统：resolveHookPermissionDecision()

**文件**：`src/services/tools/toolHooks.ts:332-400`

**功能**：合并 Hook 决策和规则系统决策。

### 4.1 决策合并算法

```typescript
// src/services/tools/toolHooks.ts:332-380
export function resolveHookPermissionDecision(
  hookResult: HookPermissionResult
): PermissionDecision {
  const { hookDecision, hookMessage, updatedInput } = hookResult

  // Hook deny：直接拒绝
  if (hookDecision === 'deny') {
    return { behavior: 'deny', decisionReason: { type: 'hook', ... }, message: hookMessage }
  }

  // Hook allow：必须通过规则系统检查
  if (hookDecision === 'allow') {
    if (requiresUserInteraction(tool)) {
      return { behavior: 'ask', decisionReason: { type: 'hook', ... }, updatedInput }
    }

    const ruleDecision = checkRuleBasedPermissions(tool, input)
    if (ruleDecision.behavior === 'deny') {
      return ruleDecision  // 规则覆盖 Hook 的 allow
    }

    return { behavior: 'allow', updatedInput, decisionReason: {...} }
  }

  // Hook ask：正常权限流程
  return { behavior: 'ask', updatedInput, decisionReason: { type: 'hook', ... } }
}
```

### 4.2 关键不变量

> **Hook 的 `allow` 不能绕过 settings.json 中的 deny 规则**

```typescript
// src/services/tools/toolExecution.ts:150
export function classifyToolError(error: unknown): string {
  if (error instanceof TelemetrySafeError) return error.telemetryMessage
  if (error instanceof Error) {
    const errnoCode = getErrnoCode(error)
    if (typeof errnoCode === 'string') return `Error:${errnoCode}`
    if (error.name && error.name !== 'Error' && error.name.length > 3)
      return error.name
    return 'Error'
  }
  return 'UnknownError'
}
```
