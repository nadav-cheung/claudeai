// src/utils/sessionStorage.ts:50
export async function appendMessageToSession(
  sessionId: string,
  message: Message
): Promise<void> {
  const sessionPath = getSessionPath(sessionId)

  // 序列化为单行 JSON
  const line = JSON.stringify(message)

  // 追加写入
  await fs.appendFile(sessionPath, line + '\n')
}

// 读取会话消息
export async function readSessionMessages(
  sessionId: string
): Promise<Message[]> {
  const sessionPath = getSessionPath(sessionId)

  if (!await fs.pathExists(sessionPath)) {
    return []
  }

  const content = await fs.readFile(sessionPath, 'utf-8')
  const lines = content.split('\n').filter(line => line.trim())

  return lines.map(line => JSON.parse(line))
}

### 2.2 @include 指令处理

```typescript
// src/utils/claudemd.ts:100
export function processIncludes(
  content: string,
  baseDir: string,
  processedFiles: Set<string> = new Set()
): string {
  // 正则匹配 @include 指令
  const includePattern = /@include\[([^\]]+)\]\((\S+)\)/g

  return content.replace(includePattern, (match, label, path) => {
    // 解析路径
    const resolvedPath = resolveIncludePath(path, baseDir)

    // 循环引用检测
    if (processedFiles.has(resolvedPath)) {
      console.warn(`Circular @include detected: ${resolvedPath}`)
      return ''
    }

    processedFiles.add(resolvedPath)

    // 读取并递归处理
    const includedContent = fs.readFileSync(resolvedPath, 'utf-8')
    return processIncludes(includedContent, pathDirname(resolvedPath), processedFiles)
  })
}

// 路径解析
function resolveIncludePath(path: string, baseDir: string): string {
  if (path.startsWith('@/')) {
    // @/absolute/path → 绝对路径
    return path.slice(2)
  }
  if (path.startsWith('@~/')) {
    // @~/home/path → home 目录
    return pathJoin(os.homedir(), path.slice(3))
  }
  if (path.startsWith('@./')) {
    // @./relative/path → 相对于 baseDir
    return pathJoin(baseDir, path.slice(3))
  }
  // @path → 相对于 baseDir
  return pathJoin(baseDir, path)
}
```typescript

export function parseClaudeMdFrontmatter(
  content: string
): { metadata: ClaudeMdMetadata; body: string } {
  const frontmatterPattern = /^---\n([\s\S]*?)\n---\n([\s\S]*)$/
  const match = content.match(frontmatterPattern)

  if (!match) {
    return { metadata: {}, body: content }
  }

  const [, frontmatter, body] = match
  const metadata = yaml.parse(frontmatter)

  return { metadata, body }
}

// src/services/SessionMemory/sessionMemory.ts:50
export function shouldExtractMemory(messages: Message[]): boolean {
  const currentTokenCount = tokenCountWithEstimation(messages)

  // 1. 检查是否达到初始化阈值
  if (!isSessionMemoryInitialized()) {
    if (!hasMetInitializationThreshold(currentTokenCount)) {
      return false
    }
    markSessionMemoryInitialized()
  }

  // 2. 检查是否达到更新阈值
  const hasMetTokenThreshold = hasMetUpdateThreshold(currentTokenCount)

  // 3. 检查工具调用次数
  const toolCallsSinceLast = countToolCallsSince(
    messages,
    lastMemoryMessageUuid
  )
  const hasMetToolCallThreshold = toolCallsSinceLast >= getToolCallsBetweenUpdates()

  // 4. 检查最后 assistant turn 是否有工具调用（安全点）
  const hasToolCallsInLastTurn = hasToolCallsInLastAssistantTurn(messages)

  // 触发条件：token 阈值 + (工具调用阈值 OR 安全点)
  return hasMetTokenThreshold &&
    (hasMetToolCallThreshold || !hasToolCallsInLastTurn)
}

## 5. 会话恢复：Session Resume

**文件**：`src/commands/resume/resume.ts:50-120`

**功能**：恢复已保存的会话继续对话。

```typescript
// src/commands/resume/resume.ts:50
export async function resumeSession(
  sessionId: string
): Promise<void> {
  // 1. 读取 JSONL 文件
  const sessionPath = getSessionPath(sessionId)
  const messages = await readSessionMessages(sessionId)

  // 2. 反序列化消息
  const deserializedMessages = messages.map(deserializeMessage)

  // 3. 处理压缩边界
  const { preBoundary, postBoundary } = splitByCompactBoundary(deserializedMessages)

  // 4. 如果有摘要，加载摘要内容
  if (postBoundary.length === 0 && preBoundary.length > 0) {
    const summary = findSummaryMessage(preBoundary)
    if (summary) {
      contextMessages = [summary, ...loadCompactKeptMessages(preBoundary)]
    }
  } else {
    contextMessages = postBoundary
  }

  // 5. 恢复工具状态
  await restoreToolState(contextMessages)

  // 6. 启动 query 循环
  await runQueryLoop(contextMessages)
}

### 练习 1：会话记忆 vs 持久记忆

**问题**：Session Memory 和 Auto Memory 有什么区别？

**答案**：

| 方面 | Session Memory | Auto Memory |
|------|---------------|--------------|
| 存储位置 | 内存或临时文件 | 持久化到 `~/.claude/memories/` |
| 生命周期 | 当前会话 | 跨会话持久化 |
| 触发时机 | 每次对话后 | 空闲时或定期 |
| 用途 | 快速恢复会话上下文 | 长期知识积累 |

```typescript
// Session Memory
const sessionMemory = await getSessionMemory(sessionId)

// Auto Memory
const memories = await scanMemoryFiles(getAutoMemDir())

async function processIncludes(claudeMdContent) {
  const lines = claudeMdContent.split('\n')
  const result = []

  for (const line of lines) {
    if (line.startsWith('@include ')) {
      const filePath = extractPath(line)
      const included = await readFile(filePath)
      result.push(included)  // 递归展开
    } else {
      result.push(line)

  }

### 练习 4：记忆提取的权限限制

**问题**：为什么 `extractAutoMemories` 需要限制工具权限？

**答案**：

**安全考虑**：
- 记忆提取运行独立的 fork agent
- 必须防止记忆提取 agent 执行危险操作
- 只允许安全的只读工具和记忆文件操作

```typescript
// 创建受限的工具权限
const restrictedCanUseTool = createAutoMemoryCanUseTool()

// 禁止：FileWrite, Bash(写入), rm 等
```text

### 练习 5：会话消息的 JSONL 格式

**问题**：会话消息为什么使用 JSONL 格式存储？

**答案**：

**JSONL（JSON Lines）优势**：
1. **追加写入**：每行一条记录，写入高效
2. **随机读取**：按行号快速定位
3. **容错性**：一行损坏不影响其他行
4. **流式处理**：可以边写边读

```jsonl
{"role":"user","content":"Hello","timestamp":1700000000}
{"role":"assistant","content":"Hi!","timestamp":1700000001}
{"type":"tool_use","name":"Read",...}
{"type":"tool_result","content":"file content",...}
```text
