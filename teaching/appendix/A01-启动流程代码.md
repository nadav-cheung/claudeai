## 1. CLI 入口：main()

**文件**：`src/main.tsx:585-700`

**功能**：CLI 主入口函数，协调整个启动流程。

```typescript
// src/main.tsx:585
export async function main() {
  // 1. 安全：防止 Windows PATH 劫持
  process.env.NoDefaultCurrentDirectoryInExePath = '1'

  // 2. 初始化警告处理
  initializeWarningHandler()

  // 3. 注册退出时的光标恢复
  process.on('exit', () => resetCursor())

  // 4. 处理 SIGINT（Ctrl+C）
  process.on('SIGINT', () => {
    // 非 print 模式下，SIGINT 正常退出
    if (!process.argv.includes('-p') && !process.argv.includes('--print')) {
      process.exit(0)
    }
  })

  // 5. 早期参数处理
  eagerLoadSettings()
  initializeEntrypoint(isNonInteractive)

  // 6. 处理 cc:// URL（Direct Connect）
  if (process.argv.some(arg => arg.startsWith('cc://'))) {
    await handleDirectConnect(process.argv)
    return
  }

  // 7. 构建 Commander.js 命令
  const program = buildCLIProgram()

  // 8. 执行命令
  await program.parseAsync(process.argv)
}
```

### 2.1 API 预连接

```typescript
// src/entrypoints/init.ts:150
async function preconnectAnthropicApi(): Promise<void> {
  const apiUrl = getAnthropicApiUrl()

  // 建立 TCP 连接
  const socket = await createConnection({
    host: apiUrl.hostname,
    port: apiUrl.port || 443,
  })

  // 预热 TLS（如果是 HTTPS）
  if (apiUrl.protocol === 'https:') {
    await tls handshake(socket, apiUrl.hostname)
  }

  // 保持连接活跃
  socket.setKeepAlive(true)

  // 存储以供后续使用
  globalApiSocket = socket
}
```

## 3. REPL 启动：launchRepl()

**文件**：`src/replLauncher.tsx:50-100`

**功能**：启动交互式 REPL 界面。

```typescript
// src/replLauncher.tsx:50
export async function launchRepl(
  root: Root,
  appProps: AppProps,
  replProps: REPLProps,
  renderAndRun: RenderFunction
): Promise<void> {
  // 1. 动态导入 React 组件
  const { App } = await import('./components/App.js')
  const { REPL } = await import('./screens/REPL.js')

  // 2. 构建 App props
  const stats = createStatsStore()
  const initialState = await buildInitialAppState()

  // 3. 构建 React 组件树
  const appElement = (
    <App
      stats={stats}
      initialState={initialState}
      getFpsMetrics={createFpsMetrics}
    >
      <FpsMetricsProvider getFpsMetrics={createFpsMetrics}>
        <StatsProvider store={stats}>
          <AppStateProvider initialState={initialState}>
            <MailboxProvider>
              <VoiceProvider>
                <REPL {...replProps} />
              </VoiceProvider>
            </MailboxProvider>
          </AppStateProvider>
        </StatsProvider>
      </FpsMetricsProvider>
    </App>
  )

  // 4. 渲染并运行
  await renderAndRun(root, appElement)
}
```
