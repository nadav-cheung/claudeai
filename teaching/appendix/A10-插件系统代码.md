## 1. Plugin 结构

```typescript
// src/plugins/types.ts
export interface Plugin {
  // 标识
  id: string                    // 唯一标识，如 "github", "slack"

  // 权限
  permissions?: {
    tools?: string[]           // 需要的工具

  }
}
```

## 2. Plugin 生命周期

**文件**：`src/services/plugins/pluginOperations.ts:50-200`

**功能**：管理 Plugin 的安装、启用、禁用、更新、卸载。

### 2.1 安装流程

```typescript
// src/services/plugins/pluginOperations.ts:50
export async function installPluginOp(
  pluginId: string,
  scope: 'user' | 'project' | 'local'
): Promise<PluginInstallationResult> {
  // 1. 在 Marketplace 中查找 Plugin
  const plugin = await searchMarketplaces(pluginId)
  if (!plugin) {
    throw new PluginNotFoundError(pluginId)
  }

  // 2. 解析依赖
  const resolvedDeps = await resolveDependencies(plugin.dependencies)

  // 3. 下载 Plugin 到缓存目录
  const pluginDir = await downloadPlugin(plugin, resolvedDeps)

  // 4. 写入 settings（声明意图）
  await updateSettingsForScope(scope, {
    enabledPlugins: { [pluginId]: true }
  })

  // 5. 记录到 installed_plugins_v2.json
  await recordPluginInstallation({
    id: pluginId,
    version: plugin.version,
    installedAt: Date.now(),
    scope,
    dir: pluginDir,
  })

  // 6. 刷新 Plugin 缓存
  await refreshPlugins()

  return { success: true, plugin }
}
```

### 2.2 启用/禁用

```typescript
// src/services/plugins/pluginOperations.ts:150
export async function setPluginEnabledOp(
  pluginId: string,
  enabled: boolean,
  scope: 'user' | 'project'
): Promise<void> {
  // 1. 解析 pluginId 和 scope
  const { pluginIdentifier, targetScope } = parsePluginId(pluginId, scope)

  // 2. 检查策略限制
  if (isBlockedByPolicy(pluginIdentifier)) {
    throw new PluginBlockedByPolicyError(pluginId)
  }

  // 3. 更新 settings
  await updateSettingsForScope(targetScope, {
    enabledPlugins: { [pluginId]: enabled }
  })

  // 4. 清除缓存
  await invalidatePluginCache(pluginId)
}
```

### 2.3 卸载流程

```typescript
// src/services/plugins/pluginOperations.ts:200
export async function uninstallPluginOp(
  pluginId: string,
  scope: 'user' | 'project'
): Promise<void> {
  // 1. 从 settings 移除
  await updateSettingsForScope(scope, {
    enabledPlugins: { [pluginId]: false }
  })

  // 2. 从 installed_plugins_v2.json 移除
  await removePluginRecord(pluginId, scope)

  // 3. 标记旧版本为孤儿
  await markPluginOrphaned(pluginId)

  // 4. 删除 Plugin 选项和密钥
  await deletePluginSecrets(pluginId)

  // 5. 可选：删除数据目录
  if (options.deleteData) {
    await deletePluginDataDir(pluginId)
  }

  // 6. 刷新缓存
  await invalidatePluginCache(pluginId)
}
```

### 3.2 加载流程

```typescript
// src/utils/plugins/loadPluginHooks.ts:50
export async function loadPluginHooks(): Promise<void> {
  // 1. 获取所有已启用的 Plugin
  const plugins = await getEnabledPlugins()

  // 2. 清空已注册的 Hooks
  clearRegisteredPluginHooks()

  // 3. 遍历每个 Plugin
  for (const plugin of plugins) {
    // 转换 Hook 配置为 Matcher
    const hookMatchers = convertPluginHooksToMatchers(plugin)

    // 注册 Hook 回调
    for (const matcher of hookMatchers) {
      registerHookCallback(matcher)
    }
  }

  // 4. 设置热重载
  setupPluginHookHotReload()
}
```

### 3.3 Hook 匹配器

```typescript
// src/utils/plugins/loadPluginHooks.ts:100
function convertPluginHooksToMatchers(
  plugin: Plugin
): PluginHookMatcher[] {
  const matchers: PluginHookMatcher[] = []

  const hookConfig = plugin.hooks
  if (!hookConfig) return matchers

  for (const [hookEvent, definitions] of Object.entries(hookConfig)) {
    for (const def of definitions) {
      matchers.push({
        event: hookEvent as HookEvent,
        matcher: createMatcher(def),
        handler: loadHookHandler(plugin, def),
        pluginRoot: plugin.root,
        pluginName: plugin.name,
      })
    }
  }

  return matchers
}

function createMatcher(def: HookDefinition): HookMatcher {
  return (context: HookContext) => {
    // 根据 Hook 类型和条件判断是否匹配
    if (def.when) {
      return evaluateCondition(def.when, context)
    }
    return true  // 无条件，始终匹配
  }
}
```

### 3.4 热重载机制

```typescript
// src/utils/plugins/loadPluginHooks.ts:150
function setupPluginHookHotReload(): void {
  // 订阅设置变更
  settingsChangeDetector.subscribe(async (changes) => {
    // 如果 policySettings 变化，需要重新评估所有 Plugin
    if (changes.policySettings) {
      const oldSnapshot = getCachedPolicySnapshot()
      const newSnapshot = await loadPolicySnapshot()

      if (!snapshotsEqual(oldSnapshot, newSnapshot)) {
        // Policy 变化，清缓存并重新加载
        await loadPluginHooks()
      }
    }
  })
}
```

## 4. Plugin 与 Skill 的关系

**文件**：`src/skills/loadSkillsDir.ts:100-200`

**功能**：Plugin 可以提供 Skill，Skill 也可以来自 Plugin。

```typescript
// src/skills/loadSkillsDir.ts:100
export async function getAllSkills(): Promise<Command[]> {
  const skills: Command[] = []

  // 1. Bundled Skills
  skills.push(...getBundledSkills())

  // 2. 文件系统 Skills（.claude/skills/）
  skills.push(...await loadSkillsFromSkillsDirs())

  // 3. MCP Skills（动态加载）
  skills.push(...getMcpSkills())

  // 4. Plugin Skills ← Plugin 提供的 Skill
  skills.push(...await getPluginSkills())

  // 5. 去重
  return deduplicateSkills(skills)
}

async function getPluginSkills(): Promise<Command[]> {
  const plugins = await getEnabledPlugins()
  const pluginSkills: Command[] = []

  for (const plugin of plugins) {
    if (plugin.skills) {
      for (const skillDef of plugin.skills) {
        // 转换为 Command 对象
        const command = convertPluginSkillToCommand(plugin, skillDef)
        pluginSkills.push(command)
      }
    }
  }

  return pluginSkills
}
```

## 练习

### 练习 1：Plugin 与 Hook 的关系

**问题**：Plugin 如何通过 Hook 扩展 Claude Code 的功能？

**答案**：

**Hook 机制**：Plugin 注册回调，事件触发时执行

```typescript
// Plugin 定义
const myPlugin = {
  name: "my-plugin",
  hooks: {
    "PostToolUse": [
      {
        when: { toolName: "Bash" },
        handler: logBashCommand
      }
    ]
  }
}
```

**Hook 执行流程**：
```
事件发生（如工具执行完成）
    ↓
Hook 系统查找匹配的 Plugin
    ↓
执行 handler 回调
    ↓
传递 context 和结果
```

**Java 对比**：类似于 Servlet Filter 或 Spring AOP 的切面通知。

---

### 练习 2：Plugin 作用域

**问题**：Plugin 的 `scope` 字段（user/project/local）有什么区别？

**答案**：

| 作用域 | 安装位置 | 可见范围 | 典型用途 |
|--------|---------|---------|---------|
| `user` | `~/.claude/plugins/` | 当前用户所有项目 | 个人工具 |
| `project` | `.claude/plugins/` | 仅当前项目 | 项目专用 |
| `local` | 临时 | 仅本次会话 | 测试/调试 |

```typescript
// scope 影响安装位置
if (scope === 'user') {
  installPath = '~/.claude/plugins/my-plugin'
} else if (scope === 'project') {
  installPath = '.claude/plugins/my-plugin'
}
```

### 练习 3：Plugin 与 Skill 的区别

**问题**：Plugin 和 Skill 有什么不同？它们如何关联？

**答案**：

| 方面 | Plugin | Skill |
|------|--------|-------|
| 功能 | 扩展系统功能 | 提供命令/工作流 |
| 生命周期 | 常驻 | 按需调用 |
| Hook 支持 | 支持 | 不支持 |
| 安装单位 | 独立包 | 单个文件/目录 |

**关系**：Plugin 可以提供 Skill
```typescript
// Plugin 提供 Skill
const myPlugin = {
  name: "my-plugin",
  skills: [
    { name: "my-skill", handler: mySkillHandler }
  ]
}
```

### 练习 4：Plugin 热重载机制

**问题**：Plugin 的热重载是如何工作的？

**答案**：

**热重载触发条件**：
```typescript
settingsChangeDetector.subscribe(async (changes) => {
  if (changes.policySettings) {
    // Policy 变化时重新加载
    await loadPluginHooks()
  }
})
```

**重载流程**：
1. 监听设置变更
2. 比较新旧 Policy snapshot
3. 如果变化，清缓存
4. 重新执行 `loadPluginHooks()`

**注意**：只重载 Hook 配置，不重载整个 Plugin 代码。

### 练习 5：Plugin 安装流程

**问题**：Plugin 安装时需要哪些步骤？

**答案**：

```typescript
async function installPlugin(pluginSpec: string) {
  // 1. 解析 Plugin 规范（npm package 或 local path）
  const { type, spec } = parsePluginSpec(pluginSpec)

  // 2. 下载/复制 Plugin 文件
  if (type === 'npm') {
    await npmInstall(spec)
  } else {
    await copyLocalPlugin(spec)
  }

  // 3. 验证 Plugin 结构
  await validatePluginStructure()

  // 4. 记录安装信息
  await recordPluginInstallation({
    id: pluginId,
    version: pluginVersion,
    installedAt: Date.now(),
    scope: 'user'
  })

  // 5. 加载 Hooks
  await loadPluginHooks()
}
```
